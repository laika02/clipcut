cmake_minimum_required(VERSION 3.24)
project(ClipCut VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Quick QuickControls2 Concurrent Test)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GST REQUIRED IMPORTED_TARGET gstreamer-1.0>=1.24 gstreamer-video-1.0 gstreamer-audio-1.0)
qt_standard_project_setup(REQUIRES 6.8)
add_library(clipcut_core src/core/Edit.cpp src/core/Edit.hpp src/export/Export.cpp src/export/Export.hpp)
target_link_libraries(clipcut_core PUBLIC Qt6::Core Qt6::Concurrent)
target_include_directories(clipcut_core PUBLIC src)
qt_add_executable(clipcut src/main.cpp src/media/Preview.cpp src/media/Preview.hpp src/media/AudioMute.cpp src/media/AudioMute.hpp src/app/PerformanceLog.cpp src/app/PerformanceLog.hpp src/app/Editor.cpp src/app/Editor.hpp src/app/FileAccess.hpp src/app/Theme.cpp src/app/Theme.hpp)
target_link_libraries(clipcut PRIVATE clipcut_core Qt6::Quick Qt6::QuickControls2 Qt6::Concurrent Qt6::Test PkgConfig::GST)
target_include_directories(clipcut PRIVATE src)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  # Resolve beside the real ELF, including when invoked through a PATH symlink.
  # RPATH covers transitive/plugin dependencies; scanners inherit LD_LIBRARY_PATH.
  set_target_properties(clipcut PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN/lib")
  target_link_options(clipcut PRIVATE "LINKER:--disable-new-dtags")
endif()
qt_add_resources(clipcut ui PREFIX / FILES qml/Main.qml qml/Timeline.qml qml/MuteRegions.qml qml/CropOverlay.qml qml/FileBrowser.qml qml/SmallButton.qml qml/CompactCheckBox.qml qml/CompactComboBox.qml qml/CompactTextField.qml qml/TimelineRuler.qml)
qt_add_resources(clipcut theme_presets PREFIX / FILES themes/dark-rose.json themes/graphite.json)
qt_add_resources(clipcut themed_controls PREFIX / FILES qml/ThemedDialog.qml)
if(WIN32)
  target_link_libraries(clipcut PRIVATE psapi)
  set_target_properties(clipcut PROPERTIES WIN32_EXECUTABLE TRUE)
  set(CLIPCUT_GST_BAD_SOURCE "" CACHE PATH "Matching gst-plugins-bad source for Qt D3D11 plugin")
  if(CLIPCUT_GST_BAD_SOURCE)
    pkg_check_modules(GSTD3D REQUIRED IMPORTED_TARGET gstreamer-d3d11-1.0)
    set(qt_sink_dir "${CLIPCUT_GST_BAD_SOURCE}/ext/qt6d3d11")
    add_library(gstqt6d3d11 MODULE
      "${qt_sink_dir}/plugin.cpp" "${qt_sink_dir}/gstqml6d3d11sink.cpp"
      "${qt_sink_dir}/gstqsg6d3d11node.cpp" "${qt_sink_dir}/gstqsg6d3d11node.h"
      "${qt_sink_dir}/gstqt6d3d11videoitem.cpp" "${qt_sink_dir}/gstqt6d3d11videoitem.h")
    target_include_directories(gstqt6d3d11 PRIVATE "${qt_sink_dir}" "${CLIPCUT_GST_BAD_SOURCE}/gst-libs")
    target_compile_definitions(gstqt6d3d11 PRIVATE GST_USE_UNSTABLE_API VERSION="1.28.8" PACKAGE="clipcut" GST_PACKAGE_NAME="ClipCut" GST_PACKAGE_ORIGIN="https://gstreamer.freedesktop.org")
    target_link_libraries(gstqt6d3d11 PRIVATE Qt6::Quick PkgConfig::GSTD3D d3d11 dxgi)
  endif()
endif()
include(CTest)
if(BUILD_TESTING)
  qt_add_executable(mute_test tests/MuteTest.cpp src/media/AudioMute.cpp src/media/AudioMute.hpp)
  target_link_libraries(mute_test PRIVATE clipcut_core Qt6::Test PkgConfig::GST)
  if(WIN32)
    set_target_properties(mute_test PROPERTIES WIN32_EXECUTABLE FALSE)
  endif()
  add_test(NAME mutes COMMAND mute_test)
  qt_add_executable(theme_test tests/ThemeTest.cpp src/app/Theme.cpp src/app/Theme.hpp)
  target_include_directories(theme_test PRIVATE src)
  target_link_libraries(theme_test PRIVATE Qt6::Gui Qt6::Test)
  qt_add_resources(theme_test test_theme_presets PREFIX / FILES themes/dark-rose.json themes/graphite.json)
  add_test(NAME themes COMMAND theme_test)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/CoreTest.cpp" AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/MediaTest.cpp")
  qt_add_executable(core_test tests/CoreTest.cpp)
  target_link_libraries(core_test PRIVATE clipcut_core Qt6::Test)
  add_test(NAME core COMMAND core_test)
  qt_add_executable(media_test tests/MediaTest.cpp)
  target_link_libraries(media_test PRIVATE clipcut_core Qt6::Core)
  if(WIN32)
    set_target_properties(core_test media_test PROPERTIES WIN32_EXECUTABLE FALSE)
  endif()
  endif()
endif()
install(TARGETS clipcut RUNTIME DESTINATION bin)
